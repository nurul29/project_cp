<script setup>
import AuthenticatedLayout from '@/Layouts/AuthenticatedLayout.vue';
import SubjectCard from '@/Components/SubjectCard.vue';
import { Head, Link } from '@inertiajs/vue3';

defineProps({
    student: Object,
    subjects: Array,
    upcomingTasks: Array,
    overallProgress: Number,
});
</script>

<template>
  <Head title="Beranda Belajar Siswa" />
  <AuthenticatedLayout>
    <!-- Hero Section with props -->
    <div class="grid grid-cols-4 gap-4">

        :subject="item" 
      />
    </div>
  </AuthenticatedLayout>
</template>

🐘 Laravel Controller & Routes Mapping (routes/web.php) 

use App\Http\Controllers\Student\DashboardController;
use App\Http\Controllers\Teacher\TeacherDashboardController;

Route::middleware(['auth', 'verified'])->group(function () {
    // Route Khusus Siswa
    Route::middleware('role:siswa')->prefix('siswa')->name('siswa.')->group(function () {
        Route::get('/dashboard', [DashboardController::class, 'index'])->name('dashboard');
        Route::get('/mapel/{subject:slug}', [SubjectController::class, 'show'])->name('subjects.show');
        Route::post('/tugas/{assignment}/submit', [AssignmentController::class, 'store'])->name('assignment.submit');
    });

    // Route Khusus Guru
    Route::middleware('role:guru')->prefix('guru')->name('guru.')->group(function () {
        Route::get('/dashboard', [TeacherDashboardController::class, 'index'])->name('dashboard');
        Route::get('/nilai', [GradebookController::class, 'index'])->name('gradebook.index');
    });
});

🚀 Langkah Cepat Instalasi di Proyek Baru:

  1. Install Laravel: laravel new sundigital --git
  2. Install Breeze dengan Vue: php artisan breeze:install vue --pest --ssr
  3. Migrasi database: php artisan migrate
  4. Jalankan frontend compiler: npm install && npm run dev
  5. Gunakan komponen Tailwind yang telah disesuaikan dengan skema warna Sundigital di atas!
